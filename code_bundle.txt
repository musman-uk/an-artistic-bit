const STORAGE_KEY = 'an_artistic_bit_state_v1';
export const FREEHAND_RESOLUTION = 256;

export class AppState {
  constructor() {
    this.gridSize = 32;
    this.lastGridResolution = 32;
    this.currentMode = 'studio'; // 'studio' | 'sandbox'
    this.currentTool = 'pencil'; // 'pencil' | 'line' | 'bucket' | 'eraser' | 'picker'
    this.currentBehavior = 'sand'; // 'sand' | 'glitch' | 'bleed' | 'wave'
    this.symmetryMode = 'off'; // 'off' | 'x' | 'y' | 'quad'
    this.currentColor = '#6366f1';
    this.simSpeed = 1.0;
    this.showGridlines = true;
    this.crtEnabled = false;
    this.audioEnabled = true;
    this.activeTheme = 'midnight'; // 'midnight' | 'gbc' | 'tv' | 'workstation'
    this.focusMode = false;

    this.grid = [];
    this.actionLog = [];
    this.undoStack = [];

    this.lineStart = null;
    this.lineEnd = null;
    this.lastDrawCoord = null;

    this.isMouseDown = false;
    this.isReplaying = false;
    this.replayTimer = null;
    this.fps = 60;
    this.lastFrameTime = performance.now();
    this.frameCount = 0;
  }

  initGrid() {
    this.grid = Array(this.gridSize).fill(null).map(() => Array(this.gridSize).fill(null));
    this.actionLog = [];
    this.undoStack = [];
    this.saveUndoState();
  }

  resizeGrid(newSize) {
    if (this.gridSize === newSize) return;

    // Save current canvas to undo stack before resizing
    this.saveUndoState();

    const oldSize = this.gridSize;
    const oldGrid = this.grid;
    const newGrid = Array(newSize).fill(null).map(() => Array(newSize).fill(null));

    // Nearest-neighbor resampling to scale pixels to the new resolution
    for (let newY = 0; newY < newSize; newY++) {
      for (let newX = 0; newX < newSize; newX++) {
        const oldX = Math.floor((newX / newSize) * oldSize);
        const oldY = Math.floor((newY / newSize) * oldSize);
        if (oldGrid[oldY] && oldGrid[oldY][oldX] !== undefined) {
          newGrid[newY][newX] = oldGrid[oldY][oldX];
        }
      }
    }

    this.gridSize = newSize;
    this.grid = newGrid;
    this.saveToStorage();
  }

  saveUndoState() {
    if (this.undoStack.length > 25) this.undoStack.shift();
    const copy = this.grid.map(row => [...row]);
    this.undoStack.push({
      grid: copy,
      gridSize: this.gridSize,
      showGridlines: this.showGridlines,
      logLength: this.actionLog.length
    });
  }

  undo() {
    if (this.undoStack.length > 0) {
      const state = this.undoStack.pop();
      this.grid = state.grid;
      if (state.gridSize) this.gridSize = state.gridSize;
      if (typeof state.showGridlines === 'boolean') this.showGridlines = state.showGridlines;
      this.actionLog = this.actionLog.slice(0, state.logLength);
      this.saveToStorage();
      return true;
    }
    return false;
  }

  clearGrid() {
    this.saveUndoState();
    this.grid = Array(this.gridSize).fill(null).map(() => Array(this.gridSize).fill(null));
    this.actionLog = [];
    this.saveToStorage();
  }

  setPixel(x, y, color) {
    if (x < 0 || x >= this.gridSize || y < 0 || y >= this.gridSize) return;
    if (this.grid[y][x] === color) return;
    this.grid[y][x] = color;
    this.actionLog.push({ x, y, color, mode: this.currentMode, timestamp: Date.now() });
  }

  saveToStorage() {
    try {
      const pixels = [];
      for (let y = 0; y < this.gridSize; y++) {
        for (let x = 0; x < this.gridSize; x++) {
          const color = this.grid[y]?.[x];
          if (color) {
            pixels.push([x, y, color]);
          }
        }
      }

      const payload = {
        gridSize: this.gridSize,
        lastGridResolution: this.lastGridResolution,
        showGridlines: this.showGridlines,
        activeTheme: this.activeTheme,
        currentColor: this.currentColor,
        symmetryMode: this.symmetryMode,
        crtEnabled: this.crtEnabled,
        audioEnabled: this.audioEnabled,
        pixels
      };

      localStorage.setItem(STORAGE_KEY, JSON.stringify(payload));
    } catch (err) {
      console.warn('Unable to auto-save canvas state:', err);
    }
  }

  loadFromStorage() {
    try {
      const raw = localStorage.getItem(STORAGE_KEY);
      if (!raw) return false;

      const data = JSON.parse(raw);
      if (!data || typeof data.gridSize !== 'number' || data.gridSize < 8 || data.gridSize > 512) {
        return false;
      }

      this.gridSize = data.gridSize;
      this.lastGridResolution = [16, 32, 48].includes(data.lastGridResolution) ? data.lastGridResolution : 32;
      this.showGridlines = typeof data.showGridlines === 'boolean' ? data.showGridlines : true;
      this.activeTheme = data.activeTheme || 'midnight';
      this.currentColor = data.currentColor || '#6366f1';
      this.symmetryMode = data.symmetryMode || 'off';
      this.crtEnabled = Boolean(data.crtEnabled);
      this.audioEnabled = typeof data.audioEnabled === 'boolean' ? data.audioEnabled : true;

      this.grid = Array(this.gridSize).fill(null).map(() => Array(this.gridSize).fill(null));
      this.actionLog = [];
      this.undoStack = [];

      if (Array.isArray(data.pixels)) {
        for (const entry of data.pixels) {
          if (!Array.isArray(entry) || entry.length < 3) continue;
          const [x, y, color] = entry;
          if (x >= 0 && x < this.gridSize && y >= 0 && y < this.gridSize && typeof color === 'string') {
            this.grid[y][x] = color;
            this.actionLog.push({ x, y, color, mode: 'studio', timestamp: Date.now() });
          }
        }
      }

      this.saveUndoState();
      return true;
    } catch (err) {
      console.warn('Unable to restore saved canvas state:', err);
      return false;
    }
  }
}

export class DrawingTools {
  static applyPixelWithSymmetry(x, y, color, state) {
    const rawTargets = [{ x, y }];

    if (state.symmetryMode === 'x' || state.symmetryMode === 'quad') {
      rawTargets.push({ x: state.gridSize - 1 - x, y });
    }
    if (state.symmetryMode === 'y' || state.symmetryMode === 'quad') {
      rawTargets.push({ x, y: state.gridSize - 1 - y });
    }
    if (state.symmetryMode === 'quad') {
      rawTargets.push({ x: state.gridSize - 1 - x, y: state.gridSize - 1 - y });
    }

    // Deduplicate target coordinates (prevents double-writing on symmetry center lines)
    const uniqueTargets = new Map();
    rawTargets.forEach(pt => {
      const key = `${pt.x},${pt.y}`;
      if (!uniqueTargets.has(key)) {
        uniqueTargets.set(key, pt);
      }
    });

    uniqueTargets.forEach(pt => state.setPixel(pt.x, pt.y, color));
  }

  static stampNib(cx, cy, color, radius, state) {
    if (radius <= 0) {
      DrawingTools.applyPixelWithSymmetry(cx, cy, color, state);
      return;
    }

    for (let dy = -radius; dy <= radius; dy++) {
      for (let dx = -radius; dx <= radius; dx++) {
        if (dx * dx + dy * dy <= radius * radius + 0.5) {
          DrawingTools.applyPixelWithSymmetry(cx + dx, cy + dy, color, state);
        }
      }
    }
  }

  static drawLine(x0, y0, x1, y1, color, state, radius = 0) {
    let currX = x0;
    let currY = y0;
    const dx = Math.abs(x1 - currX);
    const dy = Math.abs(y1 - currY);
    const sx = (currX < x1) ? 1 : -1;
    const sy = (currY < y1) ? 1 : -1;
    let err = dx - dy;

    let maxSteps = state.gridSize * 3; // Infinite loop safety bound

    while (maxSteps > 0) {
      DrawingTools.stampNib(currX, currY, color, radius, state);
      if (currX === x1 && currY === y1) break;

      const e2 = 2 * err;
      if (e2 > -dy) {
        err -= dy;
        currX += sx;
      }
      if (e2 < dx) {
        err += dx;
        currY += sy;
      }
      maxSteps--;
    }
  }

  static floodFill(startX, startY, targetColor, state) {
    if (startX < 0 || startX >= state.gridSize || startY < 0 || startY >= state.gridSize) return;

    const startColor = state.grid[startY][startX];
    if (startColor === targetColor) return;

    // Index-pointer BFS queue for fast O(n) execution even in 256x256 freehand mode
    const queue = [{ x: startX, y: startY }];
    const visited = new Uint8Array(state.gridSize * state.gridSize);
    let head = 0;

    while (head < queue.length) {
      const { x, y } = queue[head++];

      if (x < 0 || x >= state.gridSize || y < 0 || y >= state.gridSize) continue;
      const flatIdx = y * state.gridSize + x;
      if (visited[flatIdx]) continue;
      visited[flatIdx] = 1;

      if (state.grid[y][x] === startColor) {
        DrawingTools.applyPixelWithSymmetry(x, y, targetColor, state);

        queue.push({ x: x + 1, y });
        queue.push({ x: x - 1, y });
        queue.push({ x, y: y + 1 });
        queue.push({ x, y: y - 1 });
      }
    }
  }
}

export class PhysicsEngine {
  static update(state, synth) {
    if (Math.random() > state.simSpeed * 0.8) return;

    if (state.currentBehavior === 'sand') {
      this.updateSand(state, synth);
    } else if (state.currentBehavior === 'glitch') {
      this.updateGlitch(state);
    } else if (state.currentBehavior === 'bleed') {
      this.updateBleed(state);
    }
  }

  static updateSand(state, synth) {
    let moved = false;

    // Bottom-up scan to prevent double-falling in a single tick
    for (let y = state.gridSize - 2; y >= 0; y--) {
      // Alternate scan direction per row to eliminate right-side piling bias
      const xIndices = Array.from({ length: state.gridSize }, (_, i) => i);
      if (Math.random() < 0.5) xIndices.reverse();

      for (const x of xIndices) {
        const current = state.grid[y][x];
        if (!current) continue;

        // 1. Direct fall
        if (!state.grid[y + 1][x]) {
          state.grid[y + 1][x] = current;
          state.grid[y][x] = null;
          moved = true;
        } else {
          // 2. Diagonal tumble
          const leftEmpty = x > 0 && !state.grid[y + 1][x - 1];
          const rightEmpty = x < state.gridSize - 1 && !state.grid[y + 1][x + 1];

          if (leftEmpty && rightEmpty) {
            const dir = Math.random() < 0.5 ? -1 : 1;
            state.grid[y + 1][x + dir] = current;
            state.grid[y][x] = null;
            moved = true;
          } else if (leftEmpty) {
            state.grid[y + 1][x - 1] = current;
            state.grid[y][x] = null;
            moved = true;
          } else if (rightEmpty) {
            state.grid[y + 1][x + 1] = current;
            state.grid[y][x] = null;
            moved = true;
          }
        }
      }
    }

    if (moved && Math.random() < 0.15) {
      synth.playSound('sand', state.activeTheme, state.audioEnabled);
    }
  }

  static updateGlitch(state) {
    const count = Math.max(2, Math.floor((state.gridSize * state.gridSize) / 220));
    for (let i = 0; i < count; i++) {
      const rx = Math.floor(Math.random() * state.gridSize);
      const ry = Math.floor(Math.random() * state.gridSize);
      if (state.grid[ry][rx]) {
        state.grid[ry][rx] = PhysicsEngine.shiftColorHue(state.grid[ry][rx], 14);
      }
    }
  }

  static updateBleed(state) {
    const directions = [
      [0, 1],
      [0, -1],
      [1, 0],
      [-1, 0]
    ];
    const attempts = Math.max(2, Math.floor((state.gridSize * state.gridSize) / 360));

    for (let i = 0; i < attempts; i++) {
      const rx = Math.floor(Math.random() * state.gridSize);
      const ry = Math.floor(Math.random() * state.gridSize);
      const color = state.grid[ry][rx];
      if (!color) continue;

      const [dx, dy] = directions[Math.floor(Math.random() * directions.length)];
      const nx = rx + dx;
      const ny = ry + dy;

      if (nx >= 0 && nx < state.gridSize && ny >= 0 && ny < state.gridSize && !state.grid[ny][nx]) {
        state.grid[ny][nx] = color;
      }
    }
  }

  static triggerEnergyWave(centerX, centerY, state) {
    const waveColor = state.currentColor;
    const step = Math.max(1.5, state.gridSize / 22);
    const bandWidth = Math.max(0.8, state.gridSize / 40);
    let radius = step;

    if (!state.activeTimers) state.activeTimers = [];

    const interval = setInterval(() => {
      // Cancel timer if wave exits grid or user leaves sandbox mode
      if (radius > state.gridSize * 1.5 || state.currentMode !== 'sandbox') {
        clearInterval(interval);
        state.activeTimers = state.activeTimers.filter(t => t !== interval);
        return;
      }

      for (let y = 0; y < state.gridSize; y++) {
        for (let x = 0; x < state.gridSize; x++) {
          const dist = Math.hypot(x - centerX, y - centerY);
          if (Math.abs(dist - radius) < bandWidth) {
            state.grid[y][x] = waveColor;
          }
        }
      }
      radius += step;
    }, 45);

    state.activeTimers.push(interval);
  }

  static shiftColorHue(hex, degree) {
    if (!hex || !hex.startsWith('#')) return hex;

    let fullHex = hex;
    if (hex.length === 4) {
      fullHex = `#${hex[1]}${hex[1]}${hex[2]}${hex[2]}${hex[3]}${hex[3]}`;
    }

    let r = parseInt(fullHex.slice(1, 3), 16) || 0;
    let g = parseInt(fullHex.slice(3, 5), 16) || 0;
    let b = parseInt(fullHex.slice(5, 7), 16) || 0;

    r = (r + degree) % 256;
    g = (g + Math.floor(degree / 2)) % 256;
    b = (b + degree * 2) % 256;

    const toHex = (val) => Math.floor(val).toString(16).padStart(2, '0');
    return `#${toHex(r)}${toHex(g)}${toHex(b)}`;
  }
}

export class Exporter {
  static calculateStats(state) {
    let activePixels = 0;
    const colorCounts = {};

    for (let y = 0; y < state.gridSize; y++) {
      for (let x = 0; x < state.gridSize; x++) {
        const color = state.grid[y][x];
        if (color) {
          activePixels++;
          colorCounts[color] = (colorCounts[color] || 0) + 1;
        }
      }
    }

    let dominantColor = '#6366F1';
    let maxCount = 0;
    Object.entries(colorCounts).forEach(([color, count]) => {
      if (count > maxCount) {
        maxCount = count;
        dominantColor = color;
      }
    });

    let entropy = 0;
    if (activePixels > 0) {
      Object.values(colorCounts).forEach(count => {
        const p = count / activePixels;
        entropy -= p * Math.log2(p);
      });
    }

    return { activePixels, dominantColor, entropy };
  }

  static copyMatrixJSON(grid) {
    const json = JSON.stringify(grid, null, 2);
    if (navigator.clipboard && window.isSecureContext) {
      navigator.clipboard.writeText(json);
    } else {
      const temp = document.createElement('textarea');
      temp.value = json;
      document.body.appendChild(temp);
      temp.select();
      document.execCommand('copy');
      document.body.removeChild(temp);
    }
  }

  static downloadMatrixJSON(grid) {
    const json = JSON.stringify(grid, null, 2);
    const blob = new Blob([json], { type: 'application/json' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `an-artistic-bit-matrix-${Date.now()}.json`;
    a.click();
    URL.revokeObjectURL(url);
  }

    exportCanvas.width = gridSize * scale;
    exportCanvas.height = gridSize * scale;

        const color = grid[y][x];
        if (color) {
          ctx.fillStyle = color;
          ctx.fillRect(x * scale, y * scale, scale, scale);
        }
      }
    }

    const a = document.createElement('a');
    a.href = exportCanvas.toDataURL('image/png');
    a.download = `an-artistic-bit-sprite-${gridSize}x${gridSize}-${Date.now()}.png`;
    a.click();
  }

import { AppState, FREEHAND_RESOLUTION } from './state.js';
import { ThemeManager, PALETTES, THEME_TITLES } from './themes.js';
import { SynthAudio } from './synth.js';
import { DrawingTools } from './tools.js';
import { PhysicsEngine } from './physics.js';
import { Exporter } from './exporter.js';

class AnArtisticBitApp {
  constructor() {
    this.state = new AppState();
    this.synth = new SynthAudio();
    this.lastSandboxSave = 0;

    const restored = this.state.loadFromStorage();
    if (!restored) {
      this.state.initGrid();
      this.loadPalettePreset('midnight');
    } else {
      ThemeManager.applyTheme(this.state.activeTheme, this.state);
      const presetMap = { midnight: 'midnight', gbc: 'gbc', tv: 'arcade', workstation: 'monochrome' };
      this.loadPalettePreset(presetMap[this.state.activeTheme] || 'midnight', false);
      this.setColor(this.state.currentColor, false);
      this.syncRestoredUI();
    }

    this.setupActiveCanvas();
    this.setupEventListeners();

    requestAnimationFrame((t) => this.gameLoop(t));
  }

  syncRestoredUI() {
    this.setSymmetry(this.state.symmetryMode, false);
    this.updateGridResolutionUI();

    const viewport = document.getElementById('hardware-viewport');
    if (viewport) {
      viewport.classList.toggle('crt-active', this.state.crtEnabled);
    }

    const icon = document.getElementById('icon-audio');
    if (icon) {
      icon.className = this.state.audioEnabled
        ? 'fa-solid fa-volume-high text-lime-400'
        : 'fa-solid fa-volume-xmark opacity-50';
    }
  }

  setupActiveCanvas() {
    const canvasIds = {
      midnight: 'canvas-midnight',
      gbc: 'canvas-gbc',
      tv: 'canvas-tv',
      workstation: 'canvas-workstation'
    };

    const targetId = canvasIds[this.state.activeTheme] || 'canvas-midnight';
    this.canvas = document.getElementById(targetId);

    if (this.canvas) {
      this.ctx = this.canvas.getContext('2d', { alpha: true });
      this.resizeCanvas();
    }
  }

  resizeCanvas() {
    if (!this.canvas) return;
    const container = this.canvas.parentElement;
    const rect = container.getBoundingClientRect();
    const size = Math.floor(Math.min(rect.width, rect.height));

    if (size > 0) {
      this.canvas.width = size;
      this.canvas.height = size;
      this.renderCanvas();
    }
  }

  setupEventListeners() {
    window.addEventListener('resize', () => this.resizeCanvas());

    const handlePointerDown = (e) => {
      this.state.isMouseDown = true;
      this.state.lastDrawCoord = null;
      this.state.saveUndoState();

      const coords = this.getGridCoords(e);
      if (coords && this.state.currentMode === 'studio' && this.state.currentTool === 'line') {
        this.state.lineStart = coords;
        this.state.lineEnd = coords;
      } else {
        this.handlePointerAction(e);
      }
    };

    const handlePointerMove = (e) => {
      const coords = this.getGridCoords(e);
      if (coords) {
        const modeTag = this.state.showGridlines ? '' : ' (Freehand)';
        document.getElementById('status-coords').innerText = `X: ${coords.x}, Y: ${coords.y}${modeTag}`;
      }
      if (this.state.isMouseDown) {
        if (this.state.currentMode === 'studio' && this.state.currentTool === 'line') {
          this.state.lineEnd = coords;
        } else {
          this.handlePointerAction(e);
        }
      }
    };

    const handlePointerUp = (e) => {
      if (!this.state.isMouseDown) return;

      if (this.state.currentMode === 'studio' && this.state.currentTool === 'line' && this.state.lineStart) {
        const coords = this.getGridCoords(e) || this.state.lineEnd;
        if (coords) {
          const radius = this.state.showGridlines ? 0 : 1;
          DrawingTools.drawLine(
            this.state.lineStart.x,
            this.state.lineStart.y,
            coords.x,
            coords.y,
            this.state.currentColor,
            this.state,
            radius
          );
          this.synth.playSound('paint', this.state.activeTheme, this.state.audioEnabled);
        }
        this.state.lineStart = null;
        this.state.lineEnd = null;
      }

      this.state.isMouseDown = false;
      this.state.lastDrawCoord = null;
      this.state.saveToStorage();
    };

    ['canvas-midnight', 'canvas-gbc', 'canvas-tv', 'canvas-workstation'].forEach(id => {
      const cvs = document.getElementById(id);
      if (cvs) {
        cvs.addEventListener('pointerdown', handlePointerDown);
        cvs.addEventListener('pointermove', handlePointerMove);
        cvs.addEventListener('touchstart', (e) => { e.preventDefault(); handlePointerDown(e); }, { passive: false });
        cvs.addEventListener('touchmove', (e) => { e.preventDefault(); handlePointerMove(e); }, { passive: false });
      }
    });

    window.addEventListener('pointerup', handlePointerUp);
    window.addEventListener('touchend', handlePointerUp);
  }

  getGridCoords(e) {
    if (!this.canvas) return null;
    const rect = this.canvas.getBoundingClientRect();
    const clientX = e.clientX ?? e.touches?.[0]?.clientX;
    const clientY = e.clientY ?? e.touches?.[0]?.clientY;

    if (clientX === undefined || clientY === undefined) return null;

    const x = Math.floor(((clientX - rect.left) / rect.width) * this.state.gridSize);
    const y = Math.floor(((clientY - rect.top) / rect.height) * this.state.gridSize);

    if (x >= 0 && x < this.state.gridSize && y >= 0 && y < this.state.gridSize) {
      return { x, y };
    }
    return null;
  }

  handlePointerAction(e) {
    if (this.state.isReplaying) return;
    const coords = this.getGridCoords(e);
    if (!coords) {
      this.state.lastDrawCoord = null;
      return;
    }

    const { x, y } = coords;
    const isFreehand = !this.state.showGridlines;
    const prev = this.state.lastDrawCoord || coords;

    if (this.state.currentMode === 'studio') {
      if (this.state.currentTool === 'pencil') {
        const radius = isFreehand ? 1 : 0;
        DrawingTools.drawLine(prev.x, prev.y, x, y, this.state.currentColor, this.state, radius);
        this.state.lastDrawCoord = coords;
        this.synth.playSound('paint', this.state.activeTheme, this.state.audioEnabled);
      } else if (this.state.currentTool === 'eraser') {
        const radius = isFreehand ? 4 : 0;
        DrawingTools.drawLine(prev.x, prev.y, x, y, null, this.state, radius);
        this.state.lastDrawCoord = coords;
        this.synth.playSound('paint', this.state.activeTheme, this.state.audioEnabled, 150);
      } else if (this.state.currentTool === 'bucket') {
        DrawingTools.floodFill(x, y, this.state.currentColor, this.state);
        this.synth.playSound('paint', this.state.activeTheme, this.state.audioEnabled, 380);
        this.state.isMouseDown = false;
        this.state.saveToStorage();
      } else if (this.state.currentTool === 'picker') {
        const picked = this.state.grid[y][x];
        if (picked) this.setColor(picked);
        this.state.isMouseDown = false;
      }
    } else if (this.state.currentMode === 'sandbox') {
      if (['sand', 'glitch', 'bleed'].includes(this.state.currentBehavior)) {
        const radius = isFreehand ? 2 : 0;
        DrawingTools.drawLine(prev.x, prev.y, x, y, this.state.currentColor, this.state, radius);
        this.state.lastDrawCoord = coords;
        this.synth.playSound('paint', this.state.activeTheme, this.state.audioEnabled);
      } else if (this.state.currentBehavior === 'wave') {
        PhysicsEngine.triggerEnergyWave(x, y, this.state);
        this.synth.playSound('wave', this.state.activeTheme, this.state.audioEnabled);
        this.state.isMouseDown = false;
      }
    }
  }

  undo() {
    if (this.state.undo()) {
      this.updateGridResolutionUI();
      this.synth.playSound('paint', this.state.activeTheme, this.state.audioEnabled, 180);
    }
  }

  clearCanvas() {
    this.state.clearGrid();
    this.synth.playSound('clear', this.state.activeTheme, this.state.audioEnabled);
  }

  gameLoop(timestamp) {
    const dt = (timestamp - this.state.lastFrameTime) / 1000;
    this.state.lastFrameTime = timestamp;
    this.state.frameCount++;

    if (this.state.frameCount % 10 === 0 && dt > 0) {
      this.state.fps = Math.round(1 / dt);
      document.getElementById('status-fps').innerText = `FPS: ${this.state.fps}`;
    }

    if (this.state.currentMode === 'sandbox' && !this.state.isReplaying) {
      PhysicsEngine.update(this.state, this.synth);
      if (timestamp - this.lastSandboxSave > 1500) {
        this.state.saveToStorage();
        this.lastSandboxSave = timestamp;
      }
    }

    this.renderCanvas();
    requestAnimationFrame((t) => this.gameLoop(t));
  }

  renderCanvas() {
    if (!this.ctx || !this.canvas) return;
    const cellSize = this.canvas.width / this.state.gridSize;
    this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);

    for (let y = 0; y < this.state.gridSize; y++) {
      for (let x = 0; x < this.state.gridSize; x++) {
        const color = this.state.grid[y][x];
        if (color) {
          this.ctx.fillStyle = color;
          this.ctx.fillRect(x * cellSize, y * cellSize, Math.ceil(cellSize), Math.ceil(cellSize));
        }
      }
    }

    if (this.state.currentMode === 'studio' && this.state.currentTool === 'line' && this.state.lineStart && this.state.lineEnd) {
      this.ctx.strokeStyle = this.state.currentColor;
      this.ctx.lineWidth = this.state.showGridlines ? Math.max(1, cellSize * 0.8) : 2.5;
      this.ctx.lineCap = 'round';
      this.ctx.beginPath();
      this.ctx.moveTo((this.state.lineStart.x + 0.5) * cellSize, (this.state.lineStart.y + 0.5) * cellSize);
      this.ctx.lineTo((this.state.lineEnd.x + 0.5) * cellSize, (this.state.lineEnd.y + 0.5) * cellSize);
      this.ctx.stroke();
    }

    if (this.state.showGridlines) {
      this.ctx.strokeStyle = 'rgba(255, 255, 255, 0.08)';
      this.ctx.lineWidth = 1;

      for (let i = 0; i <= this.state.gridSize; i++) {
        this.ctx.beginPath();
        this.ctx.moveTo(i * cellSize, 0);
        this.ctx.lineTo(i * cellSize, this.canvas.height);
        this.ctx.stroke();

        this.ctx.beginPath();
        this.ctx.moveTo(0, i * cellSize);
        this.ctx.lineTo(this.canvas.width, i * cellSize);
        this.ctx.stroke();
      }
    }

    const stats = Exporter.calculateStats(this.state);
    document.getElementById('status-pixels').innerText = `Active Pixels: ${stats.activePixels}`;
    document.getElementById('status-entropy').innerText = `Entropy: ${stats.entropy.toFixed(2)} bits`;
  }

  startReplay() {
    if (this.state.actionLog.length === 0) return;
    this.state.isReplaying = true;
    document.getElementById('overlay-replay').classList.remove('hidden');

    this.state.grid = Array(this.state.gridSize).fill(null).map(() => Array(this.state.gridSize).fill(null));
    let index = 0;
    const stepStride = Math.max(1, Math.floor(this.state.actionLog.length / 450));
    this.synth.playSound('clear', this.state.activeTheme, this.state.audioEnabled);

    this.state.replayTimer = setInterval(() => {
      if (index >= this.state.actionLog.length) {
        this.stopReplay();
        return;
      }

      for (let s = 0; s < stepStride && index < this.state.actionLog.length; s++, index++) {
        const action = this.state.actionLog[index];
        if (this.state.grid[action.y]) {
          this.state.grid[action.y][action.x] = action.color;
        }
      }

      if (index % 3 === 0) {
        this.synth.playSound('paint', this.state.activeTheme, this.state.audioEnabled, 280 + (index % 12) * 15);
      }
    }, 18);
  }

  stopReplay() {
    this.state.isReplaying = false;
    clearInterval(this.state.replayTimer);
    document.getElementById('overlay-replay').classList.add('hidden');
  }

    else if (themeKey === 'workstation') this.loadPalettePreset('monochrome');

    this.state.saveToStorage();

  }

    [16, 32, 48].forEach(s => {

      freehandBadge.classList.toggle('hidden', !isFreehand);

      if ([16, 32, 48].includes(this.state.gridSize)) {
        this.state.lastGridResolution = this.state.gridSize;

      this.state.resizeGrid(FREEHAND_RESOLUTION);
    } else {
      // Returning to Grid mode: snap back to last selected grid resolution
      const targetGrid = this.state.lastGridResolution || 32;
      this.state.resizeGrid(targetGrid);
    }

    this.updateGridResolutionUI();
    this.resizeCanvas();
  }

    document.getElementById('souvenir-res').innerText = modeLabel;
    document.getElementById('souvenir-dominant-hex').innerText = stats.dominantColor.toUpperCase();

  }

  downloadSouvenirCard() {
    const stats = Exporter.calculateStats(this.state);
    const housingTitle = THEME_TITLES[this.state.activeTheme] || 'Midnight Studio';
    Exporter.renderSouvenirCard(
      this.canvas,
      housingTitle,
      stats.activePixels,
      stats.entropy,
      this.state.gridSize,
      this.state.showGridlines
    );
  }
